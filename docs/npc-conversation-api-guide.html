整合性

同時操作を直列化

送信・振り返り時に会話を排他ロック。expectedTurn で古い画面からの重複送信も検出します。

失敗耐性

成功した往復だけ保存

LLM失敗時はユーザー発言も保存されません。入力を残して再送できるUIが適します。

冪等性

振り返りは再取得可能

REVIEWED後の再呼び出しは保存済み結果を返し、EXPや話題を二重反映しません。

type ProblemDetails = { code?: string; detail?: string };

export class ApiError extends Error {
  constructor(
    readonly status: number,
    readonly code?: string,
    detail?: string,
  ) {
    super(detail ?? "APIリクエストに失敗しました。");
  }
}

function readCookie(name: string) {
  const prefix = `${name}=`;
  const item = document.cookie
    .split("; ")
    .find((value) => value.startsWith(prefix));
  return item ? decodeURIComponent(item.slice(prefix.length)) : null;
}

async function csrfToken() {
  const response = await fetch("/api/auth/csrf", {
    credentials: "same-origin",
  });
  if (!response.ok) throw new ApiError(response.status);

  const token = readCookie("XSRF-TOKEN");
  if (!token) throw new ApiError(0, "CSRF_TOKEN_NOT_FOUND");
  return token;
}

async function request<T>(url: string, init: RequestInit = {}) {
  const method = (init.method ?? "GET").toUpperCase();
  const headers = new Headers(init.headers);

  if (init.body) headers.set("Content-Type", "application/json");
  if (method !== "GET" && method !== "HEAD") {
    headers.set("X-XSRF-TOKEN", await csrfToken());
  }

  const response = await fetch(url, {
    ...init,
    method,
    headers,
    credentials: "same-origin",
    cache: "no-store",
  });

  if (!response.ok) {
    const problem = (await response.json().catch(() => ({}))) as ProblemDetails;
    throw new ApiError(response.status, problem.code, problem.detail);
  }
  return (await response.json()) as T;
}

export const birthApi = {
  getNpc: () => request<Npc>("/api/npc"),

  createNpc: (body: { presetId: string; name: string }) =>
    request<Npc>("/api/npc", {
      method: "POST",
      body: JSON.stringify(body),
    }),

  start: () =>
    request<Conversation>("/api/conversations", {
      method: "POST",
      body: JSON.stringify({ type: "BIRTH", scene: null }),
    }),

  getConversation: (id: string) =>
    request<Conversation>(`/api/conversations/${id}`),

  send: (id: string, text: string, expectedTurn: number) =>
    request<SendMessageResult>(`/api/conversations/${id}/messages`, {
      method: "POST",
      body: JSON.stringify({ text, expectedTurn }),
    }),

  review: (id: string) =>
    request<ReviewResult>(`/api/conversations/${id}/review`, {
      method: "POST",
    }),
};

src/hooks/use-birth-conversation.ts
"use client";

import { useCallback, useEffect, useState } from "react";
import {
  ApiError,
  birthApi,
  type Conversation,
  type Npc,
  type ReviewResult,
} from "@/lib/npc-conversation-api";

type Phase = "loading" | "setup" | "conversation" | "complete";

export function useBirthConversation() {
  const [phase, setPhase] = useState<Phase>("loading");
  const [npc, setNpc] = useState<Npc | null>(null);
  const [conversation, setConversation] =
    useState<Conversation | null>(null);
  const [review, setReview] = useState<ReviewResult | null>(null);
  const [pending, setPending] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const load = useCallback(async () => {
    setPhase("loading");
    setError(null);
    try {
      const currentNpc = await birthApi.getNpc();
      setNpc(currentNpc);

      if (currentNpc.bornAt) {
        setPhase("complete");
        return;
      }

      const currentConversation = await birthApi.start();
      setConversation(currentConversation);
      setPhase("conversation");
    } catch (cause) {
      if (cause instanceof ApiError && cause.code === "NPC_NOT_FOUND") {
        setPhase("setup");
        return;
      }
      setError(toMessage(cause));
    }
  }, []);

  useEffect(() => {
    void load();
  }, [load]);

  async function createNpc(presetId: string, name: string) {
    if (pending) return;
    setPending(true);
    setError(null);
    try {
      const created = await birthApi.createNpc({
        presetId,
        name: name.trim(),
      });
      const started = await birthApi.start();
      setNpc(created);
      setConversation(started);
      setPhase("conversation");
    } catch (cause) {
      setError(toMessage(cause));
    } finally {
      setPending(false);
    }
  }

  async function send(text: string) {
    if (!conversation || pending || conversation.status !== "IN_PROGRESS") {
      return false;
    }

    setPending(true);
    setError(null);
    try {
      const result = await birthApi.send(
        conversation.id,
        text.trim(),
        conversation.turn,
      );
      setConversation({
        ...conversation,
        turn: result.turn,
        canFinish: result.canFinish,
        status: result.finished ? "FINISHED" : "IN_PROGRESS",
        messages: [
          ...conversation.messages,
          { role: "USER", text: text.trim() },
          result.reply,
        ],
      });
      return true;
    } catch (cause) {
      if (
        cause instanceof ApiError &&
        (cause.code === "TURN_MISMATCH" ||
          cause.code === "CONVERSATION_FINISHED")
      ) {
        setConversation(await birthApi.getConversation(conversation.id));
        setError("会話を最新の状態へ更新しました。");
      } else {
        setError(toMessage(cause));
      }
      return false;
    } finally {
      setPending(false);
    }
  }

  async function finish() {
    if (!conversation?.canFinish || pending) return;
    setPending(true);
    setError(null);
    try {
      const result = await birthApi.review(conversation.id);
      setReview(result);
      setConversation({ ...conversation, status: "REVIEWED" });
      setPhase("complete");
    } catch (cause) {
      setError(toMessage(cause));
    } finally {
      setPending(false);
    }
  }

  return {
    phase,
    npc,
    conversation,
    review,
    pending,
    error,
    reload: load,
    createNpc,
    send,
    finish,
  };
}

function toMessage(cause: unknown) {
  if (!(cause instanceof ApiError)) {
    return "サーバーに接続できませんでした。";
  }
  if (cause.code === "LLM_CALL_FAILED") {
    return "AIの返事を受け取れませんでした。入力を残して再試行できます。";
  }
  if (cause.code === "DEMO_LIMIT_EXCEEDED") {
    return "デモ上限に達しました。AI利用設定を変更してください。";
  }
  if (cause.status === 401) return "再ログインしてください。";
  return cause.message;
}

              onChange={(event) => setName(event.target.value)}
            />
          </label>
          {error && <div className="alert alert-error" role="alert">{error}</div>}
          <button className="btn btn-primary" disabled={pending || !name.trim()}>
            この分身と話す
          </button>
        </div>
      </form>
    );
  }

  if (phase === "complete") {
    return (
      <section className="mx-auto mt-16 max-w-xl text-center">
        <LuSparkles className="mx-auto size-12 text-primary" aria-hidden="true" />
        <h1 className="mt-4 text-3xl font-bold">{npc?.name ?? "分身"}が誕生しました</h1>
        {review && <p className="mt-4">{review.feedback}</p>}
        <Link className="btn btn-primary mt-6" href="/home">
          <LuHouse aria-hidden="true" />
          家へ行く
        </Link>
      </section>
    );
  }

  if (!npc || !conversation) return null;

  function onSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const text = draft.trim();
    if (!text) return;
    void send(text).then((sent) => {
      if (sent) setDraft("");
    });
  }

  const finished = conversation.status !== "IN_PROGRESS";

  return (
    <section className="mx-auto flex min-h-screen max-w-4xl flex-col p-5">
      <header className="flex items-center justify-between border-b border-base-300 pb-4">
        <div>
          <h1 className="text-xl font-bold">{npc.name}との最初の会話</h1>
          <p className="text-sm text-base-content/60">
            あとから答え直す必要はありません。自然に話してください。
          </p>
        </div>
        <span className="badge badge-ghost">{conversation.turn}/6 往復</span>
      </header>

      <div className="flex-1 space-y-4 py-6" aria-live="polite">
        {conversation.messages.map((message, index) => {
          const user = message.role === "USER";
          return (
            <div key={index} className={`chat ${user ? "chat-end" : "chat-start"}`}>
              <div className="chat-header">{user ? "あなた" : npc.name}</div>
              <div className={`chat-bubble ${user ? "chat-bubble-primary" : ""}`}>
                {message.text}
              </div>
            </div>
          );
        })}

        {pending && !finished && (
          <div className="chat chat-start" aria-label={`${npc.name}が考え中`}>
            <div className="chat-bubble">
              <LuLoaderCircle className="animate-spin" aria-hidden="true" />
            </div>
          </div>
        )}
      </div>

      {error && <div className="alert alert-error mb-3" role="alert">{error}</div>}

      {!finished && (
        <form className="flex gap-2" onSubmit={onSubmit}>
          <input
            className="input input-bordered flex-1"
            value={draft}
            maxLength={2000}
            disabled={pending}
            placeholder={`${npc.name}に返事をする`}
            onChange={(event) => setDraft(event.target.value)}
          />
          <button className="btn btn-primary" disabled={pending || !draft.trim()}>
            <LuSend aria-hidden="true" />
            送信
          </button>
        </form>
      )}

      {conversation.canFinish && (
        <button className="btn btn-secondary mt-3" disabled={pending} onClick={finish}>
          <LuSparkles aria-hidden="true" />
          {finished ? "振り返って誕生を確定する" : "ここまでの会話で誕生を確定する"}
        </button>
      )}
    </section>
  );
}

フロント大枠を作るときの要点

デザインを自由に変えてよい部分と、API契約として維持する部分を分けています。

自由に変更しやすい

体験・見た目

  • プリセット選択のレイアウト
  • 誕生演出・アニメーション
  • チャット吹き出しやNPC配置
  • 進捗表示の見せ方
  • 完了画面の情報設計
維持すべき契約

状態・安全性

  • 変更前のCSRF取得とヘッダー送信
  • セッションCookieを同一オリジンで送る
  • expectedTurnを毎回送る
  • TURN_MISMATCH時はサーバ状態へ同期
  • canFinish/statusをボタン制御に使う

最小実装順

  1. GET /api/npcで初期分岐
  2. POST /api/npcで名前・見た目確定
  3. POST /api/conversationsで開始/再開
  4. GET conversationで履歴描画
  5. POST messagesで1往復ずつ進行
  6. POST reviewで誕生確定
  7. 完了後GET /api/user/meまたは画面refresh

先に判断したい仕様差分

  • プリセットID: バックエンドでも固定7種を検証するか
  • 文字数: 名前30文字をUI上どう数えるか
  • 早期終了: 3往復後のボタン文言と演出
  • LLM待機: 入力ロック、再試行、タイムアウト表現
  • 将来共通化: BIRTH専用画面から共通チャット殻を切り出すか
現時点の検証境界: NPC・会話機能に専用の自動テストは見当たりません。資料はソースコードと要件の静的確認に基づき、実DB・LLMプロバイダーを通したE2E動作までは確認していません。
